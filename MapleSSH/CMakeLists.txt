cmake_minimum_required(VERSION 3.24)
cmake_policy(SET CMP0091 NEW)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(MapleSSH VERSION 0.1.0 LANGUAGES C CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "MapleSSH requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 11)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
option(MAPLE_DEPS_ONLY "Build the UI and terminal dependencies only" OFF)
option(MAPLE_BUILD_GUI "Build the desktop executable" ON)
set(MAPLE_WX_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/out/deps/wxWidgets-3.2.11" CACHE PATH "Path to a complete wxWidgets source distribution")
set(MAPLE_DOWNLOADS "${CMAKE_CURRENT_SOURCE_DIR}/out/downloads" CACHE PATH "Verified source archive directory")
set(OPENSSL_USE_STATIC_LIBS TRUE)
set(OPENSSL_MSVC_STATIC_RT TRUE)
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/utf-8> $<$<COMPILE_LANGUAGE:C>:/utf-8>)
add_compile_definitions(UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00)
include(cmake/dependencies.cmake)
enable_testing()
add_executable(maplessh_terminal_tests tests/terminal_tests.cpp)
target_link_libraries(maplessh_terminal_tests PRIVATE maple_vterm)
add_test(NAME terminal_sequences COMMAND maplessh_terminal_tests)
if(MAPLE_DEPS_ONLY)
    return()
endif()
find_package(OpenSSL 3.5 REQUIRED COMPONENTS Crypto)
set(CRYPTO_BACKEND OpenSSL CACHE STRING "" FORCE)
set(ENABLE_ZLIB_COMPRESSION OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_STATIC_LIBS ON CACHE BOOL "" FORCE)
add_subdirectory("${MAPLE_DOWNLOADS}/libssh2-1.11.1" "${CMAKE_BINARY_DIR}/libssh2" EXCLUDE_FROM_ALL)
add_library(maplessh_core STATIC src/model.cpp src/crypto.cpp src/storage.cpp src/host_keys.cpp src/webdav.cpp src/backup_labels.cpp src/command_input.cpp)
target_include_directories(maplessh_core PUBLIC src "${MAPLE_DOWNLOADS}/include")
target_link_libraries(maplessh_core PUBLIC OpenSSL::Crypto winhttp ws2_32 crypt32 xmllite shlwapi ole32)
target_compile_options(maplessh_core PRIVATE /W4)
if(MAPLE_BUILD_GUI)
include(cmake/notices.cmake)
add_executable(MapleSSH WIN32 src/app.cpp src/main_frame.cpp src/dialogs.cpp src/terminal.cpp src/ssh_client.cpp resources/app.rc resources/app.manifest)
target_link_libraries(MapleSSH PRIVATE maplessh_core maple_vterm libssh2_static wx::core wx::base wx::aui wx::xml)
target_include_directories(MapleSSH PRIVATE src resources "${CMAKE_BINARY_DIR}")
target_compile_options(MapleSSH PRIVATE /W4)
set_target_properties(MapleSSH PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endif()
include(CTest)
enable_testing()
add_executable(maplessh_tests tests/core_tests.cpp)
target_link_libraries(maplessh_tests PRIVATE maplessh_core)
foreach(test_case model_roundtrip backup_whitelist invalid_models crypto_roundtrip crypto_rejects_damage atomic_storage passwordless_portability legacy_storage_preserved)
    add_test(NAME ${test_case} COMMAND maplessh_tests ${test_case})
endforeach()
add_executable(maplessh_network_tests tests/network_tests.cpp)
add_executable(maplessh_command_tests tests/command_input_tests.cpp)
target_link_libraries(maplessh_command_tests PRIVATE maplessh_core)
add_test(NAME multiline_command_input COMMAND maplessh_command_tests)
add_executable(maplessh_quick_commands_tests tests/quick_commands_tests.cpp)
target_link_libraries(maplessh_quick_commands_tests PRIVATE maplessh_core)
add_test(NAME quick_commands_backup COMMAND maplessh_quick_commands_tests)
add_executable(maplessh_credentials_tests tests/credentials_tests.cpp)
target_link_libraries(maplessh_credentials_tests PRIVATE maplessh_core)
add_test(NAME reusable_credentials_and_groups COMMAND maplessh_credentials_tests)
target_link_libraries(maplessh_network_tests PRIVATE maplessh_core)
add_test(NAME url_and_host_trust COMMAND maplessh_network_tests)
add_executable(maplessh_integration_tests tests/integration_tests.cpp src/ssh_client.cpp)
target_link_libraries(maplessh_integration_tests PRIVATE maplessh_core libssh2_static)
